def agent_metrics(group):
    from evaluation.intent_eval import nearest_rank
    totals={}
    for name,dataset,prefix,seed in GROUPS[group]:
        totals[name]={}
        for mode in ['rule','hybrid']:
            run=ROOT/'evaluation/results'/f'{prefix}_{mode}'
            cases={c['id']:c for c in rows(run/'cases.jsonl')}
            reviews={r['id']:r for r in rows(run/'reviews.jsonl')}
            raw=rows(run/'results.jsonl')
            turn_rows=[]
            for result in raw:
                case=cases[result['id']]
                for index,expected in enumerate(case['turns']):
                    turn=result.get('turns',[])[index] if index<len(result.get('turns',[])) else {}
                    review=reviews[result['id']]['turn_reviews'][index]
                    events=(turn.get('runtime') or {}).get('events',[])
                    intent=[e for e in events if e['event']=='intent_resolved']
                    required=expected.get('required_tools',[])
                    report=bool(expected.get('gold_intent')) or 'fetch_external_data' in required or bool(intent)
                    turn_rows.append({'id':result['id'],'turn_index':index+1,'query':expected['query'],
                        'answer_score':review['answer_score'],'grounded':review['grounded'],'note':review['note'],
                        'task_pass':reviews[result['id']]['task_pass'],'status':turn.get('status','not_run'),
                        'tool_selection':turn.get('tool_selection'),'report_turn':report,
                        'intent_latency_ms':intent[-1]['latency_ms'] if intent else None,
                        'intent_mode':intent[-1]['mode'] if intent else None,
                        'intent_extra_model_calls':sum(e['event']=='call_started' and e.get('kind')=='model' and e.get('name')=='agent.intent' for e in events),
                        'intent_budget_skip':int(bool(intent and intent[-1].get('fallback_reason') in ['model_budget','time_budget','context_budget'])),
                        'intent_fallback_reason':intent[-1].get('fallback_reason') if intent else None,
                        'intent_conflicts':intent[-1].get('conflicts',[]) if intent else [],
                        'result_sha256':reviews[result['id']]['result_sha256']})
            jsonl(run/'p3_turn_metrics.jsonl',turn_rows)
            report_rows=[t for t in turn_rows if t['report_turn']]
            latencies=[t['intent_latency_ms']/1000 for t in report_rows if t['intent_latency_ms'] is not None]
            totals[name][mode]={'planned_tasks':len(cases),'planned_turns':len(turn_rows),
                'whole_task_pass':sum(r['task_pass'] for r in reviews.values()),
                'content_score_counts':{str(s):sum(t['answer_score']==s for t in turn_rows) for s in [0,1,2]},
                'faithful_turns':sum(t['grounded'] for t in turn_rows),
                'report_turns':len(report_rows),'intent_observed_turns':len(latencies),
                'intent_latency_p50_seconds':nearest_rank(latencies,.50),'intent_latency_p95_seconds':nearest_rank(latencies,.95),
                'extra_intent_model_calls':sum(t['intent_extra_model_calls'] for t in turn_rows),
                'budget_skips':sum(t['intent_budget_skip'] for t in turn_rows),
                'rows':turn_rows,'evidence':str(run.relative_to(ROOT))}
        before={ (t['id'],t['turn_index']):t for t in totals[name]['rule']['rows']}
        increments=[]
        for t in totals[name]['hybrid']['rows']:
            b=before[(t['id'],t['turn_index'])]
            if t['intent_latency_ms'] is not None and b['intent_latency_ms'] is not None:
                increments.append((t['intent_latency_ms']-b['intent_latency_ms'])/1000)
        paired=json.loads((ROOT/'evaluation/results'/f'{prefix}_blind/paired_summary.json').read_text(encoding='utf-8'))
        totals[name]['paired']=paired
        totals[name]['paired_intent_latency_increment']={'count':len(increments),'p50_seconds':nearest_rank(increments,.50),'p95_seconds':nearest_rank(increments,.95)}
    dest=ROOT/'evaluation/results'/f'intent_p3_{group}_analysis'
    dest.mkdir(exist_ok=True)
    dump(dest/'summary.json',totals)
    return totals

def review_view(group,name,start,end):
    spec=next(s for s in GROUPS[group] if s[0]==name)
    package=rows(ROOT/'evaluation/results'/f'{spec[2]}_blind/review_package.jsonl')
    for sample in package[start-1:end]:
        print(json.dumps({'sample_id':sample['sample_id'],'case':sample['case'],'result_sha256':sample['result_sha256'],
            'execution_status':sample['execution_status'],'turn_execution_outcomes':sample['turn_execution_outcomes'],
            'turns':sample['turns'],'tool_results':sample['tool_results'],'rag_answers':sample.get('rag_answers'),
            'rag_input_contexts':[r.get('rag_contexts') for r in sample['model_requests']]},ensure_ascii=False))

def component_summary():
    from evaluation.intent_eval import build_case_inputs,adapt_result,score_case,_case_gold
    from agent.intent.schema import IntentProposal
    from agent.intent.resolve import resolve_turn_intent
    from agent.runtime import RunContext
    from utils.records import load_records
    from scripts.seal_evaluation_run import seal_run,result_sha256
    dest=ROOT/'evaluation/results/intent_p3_A_analysis'
    dest.mkdir(exist_ok=False)
    casepath=ROOT/'evaluation/intent_v1_r1/test.jsonl'
    cases={c['id']:c for c in rows(casepath)}
    records=load_records(str(ROOT/'data/external/records.csv'))
    mode_summaries={}
    results={}
    for mode,count in [('rule',1),('llm_only',3),('hybrid',3)]:
        summaries=[]
        for idx in range(1,count+1):
            run=ROOT/'evaluation/results'/f'intent_v1_r1_test_p3_{mode}_r{idx:03d}'
            seal_run(run,verify=True)
            summary=json.loads((run/'summary.json').read_text(encoding='utf-8'))['aggregate']
            summaries.append({'repeat':f'r{idx:03d}','evidence':str(run.relative_to(ROOT)),**summary})
            results[(mode,idx)]={r['case_id']:r for r in rows(run/'results.jsonl')}
        mode_summaries[mode]={'runs':summaries,
            'mean_exact_accuracy':statistics.mean(s['exact_accuracy'] for s in summaries),
            'worst_exact_accuracy':min(s['exact_accuracy'] for s in summaries),
            'mean_dangerous':statistics.mean(s['severity_counts']['dangerous'] for s in summaries),
            'worst_dangerous':max(s['severity_counts']['dangerous'] for s in summaries),
            'mean_slot_accuracy':{k:statistics.mean(s['slot_accuracy'][k] for s in summaries) for k in summaries[0]['slot_accuracy']},
            'worst_slot_accuracy':{k:min(s['slot_accuracy'][k] for s in summaries) for k in summaries[0]['slot_accuracy']},
            'all_conflicts':dict(sum((Counter(s['conflict_counts']) for s in summaries),Counter()))}
    # Replay only the stored proposal through the unchanged deterministic
    # llm_only merge, replacing its parser with a constant. No model is called,
    # no benchmark is rerun, and no replay timing enters measured costs.
    validation_effects=[]
    observed_pairs=[]
    for idx in range(1,4):
        for case_id,h in results[('hybrid',idx)].items():
            case=cases[case_id]
            raw=h['raw_proposal']
            before=None
            if raw is not None:
                inputs=build_case_inputs(case,records)
                with patch('agent.intent.resolve.parse_turn_intent',return_value=IntentProposal.model_validate(raw)):
                    run=RunContext()
                    output=resolve_turn_intent(**inputs,mode='llm_only',run=run)
                if run.counts.get('model',0): raise RuntimeError('Offline proposal replay called a model')
                prediction=adapt_result(output,inputs['known_fields'])
                before=score_case(prediction,_case_gold(case),case.get('unknown_aliases',()),status='ok')
                dangerous_removed='dangerous' in before['error_flags'] and 'dangerous' not in h['score']['error_flags']
                validation_worsened=before['pass'] and not h['score']['pass']
                if dangerous_removed or validation_worsened:
                    validation_effects.append({'case_id':case_id,'repeat':f'r{idx:03d}',
                        'query':case['query'],'raw_hybrid_proposal_ablation_prediction':prediction,
                        'raw_hybrid_proposal_ablation_score':before,'final_hybrid_prediction':h['prediction'],'final_hybrid_score':h['score'],
                        'conflicts':h['trace'].get('conflicts',[]),'dangerous_removed':dangerous_removed,
                        'validation_worsened':validation_worsened,'hybrid_result_sha256':result_sha256(h),
                        'evidence':f'evaluation/results/intent_v1_r1_test_p3_hybrid_r{idx:03d}/results.jsonl',
                        'method':'Deterministic stored-proposal replay; zero inference calls; used for attribution only.'})
            l=results[('llm_only',idx)][case_id]
            if 'dangerous' in l['score']['error_flags'] and 'dangerous' not in h['score']['error_flags']:
                observed_pairs.append({'case_id':case_id,'repeat':f'r{idx:03d}','query':case['query'],
                    'llm_only_prediction':l['prediction'],'hybrid_prediction':h['prediction'],
                    'hybrid_conflicts':h['trace'].get('conflicts',[]),
                    'same_hybrid_raw_proposal_dangerous':bool(before and 'dangerous' in before['error_flags']),
                    'llm_only_result_sha256':result_sha256(l),'hybrid_result_sha256':result_sha256(h),
                    'hybrid_fully_correct':h['score']['pass']})
    dump(dest/'summary.json',mode_summaries)
    jsonl(dest/'validation_effects.jsonl',validation_effects)
    jsonl(dest/'llm_dangerous_hybrid_safe_pairs.jsonl',observed_pairs)
    examples=[]
    for case_id,label,note in [
        ('IT001','成功：同义字段和顺序','将“主刷还能撑几天”和“平均每天清扫的面积”映射为目录字段，顺序、范围和月份均正确。'),
        ('IT021','成功：区域限定保护','V4拒绝将“厨房覆盖率”映射为通用“覆盖率”，保留未知原指标，未选择目录里的近似字段。'),
        ('IT047','成功：明确查询锚点的月份算术','以当前查询2025-08为明确起点，代码重算前一个月为2025-07；组件选择正确不授予该未查询月的数字输出权限。'),
        ('IT041','失败：日期被误当字段限定词','原始mention携带日期，V4把日期当作字段限定，误将可答的“玩具避障率”降为未知，造成safe_fail。'),
        ('IT044','失败：错误锚点声明被重算放大','gold应由上轮2025-07得到2025-08；模型的锚点声明使V7重算为2025-09，产生错误但合法的目标月。校验不保证语义锚点判断正确。'),
        ('IT075','失败：撤回月份和字段误选未拦住','应选择2025-05主刷寿命，最终选择2025-04覆盖率；无冲突码。月份有据和目录成员检查不足以证明当前肯定意图角色。')]:
        c=cases[case_id]; r=results[('hybrid',1)][case_id]
        examples.append({'case_id':case_id,'label':label,'query':c['query'],'history':c.get('history',[]),
            'final_intent':{**r['final_intent'],'scope':r['final_scope'],'knowledge_requested':r['prediction']['knowledge_requested'],'excluded_fields':r['final_excluded_fields']},
            'conflicts':r['trace']['conflicts'],'note':note,'result_sha256':result_sha256(r),
            'evidence':'evaluation/results/intent_v1_r1_test_p3_hybrid_r001/results.jsonl'})
    dump(dest/'examples.json',examples)
    lines=['# P3 A：组件 test 正式评测','',
        '冻结运行时：`c105253026453bb203aab1186d68aae0bb10acd3`。作者已批准 d61a06a 冻结的金标准与本批次。每次100题；rule×1、llm_only×3、hybrid×3，共700条；未重跑、未筛选。所有原始状态留在分母。',
        '', '| 模式/重复 | 完全正确/100 | 危险错误 | wrong_scope | safe_fail | other_error | 执行失败 | 回退 | p50/p95 秒（全部状态） |',
        '|---|---:|---:|---:|---:|---:|---:|---:|---:|']
    for mode,s in mode_summaries.items():
        for r in s['runs']:
            c=r['severity_counts']; l=r['latency_seconds']['all']
            lines.append(f"| {mode}/{r['repeat']} | {r['exact_count']} | {c['dangerous']} | {c['wrong_scope']} | {c['safe_fail']} | {c['other_error']} | {r['execution_failure_count']} | {r['fallback_count']} | {l['p50']:.4f}/{l['p95']:.4f} |")
    lines+=['','| 模式 | 正确率均值/最差 | 危险错误均值/最差（最大） |','|---|---:|---:|']
    for mode,s in mode_summaries.items(): lines.append(f"| {mode} | {s['mean_exact_accuracy']:.1%}/{s['worst_exact_accuracy']:.1%} | {s['mean_dangerous']:.2f}/{s['worst_dangerous']} |")
    lines+=['','| 槽位 | rule | llm_only 均值/最差 | hybrid 均值/最差 |','|---|---:|---:|---:|']
    for slot in mode_summaries['rule']['mean_slot_accuracy']:
        vals=[f"{mode_summaries[m]['mean_slot_accuracy'][slot]:.1%}/{mode_summaries[m]['worst_slot_accuracy'][slot]:.1%}" for m in ['llm_only','hybrid']]
        lines.append(f"| {slot} | {mode_summaries['rule']['mean_slot_accuracy'][slot]:.1%} | {vals[0]} | {vals[1]} |")
    lines+=['','全部冲突码计数（跨完整重复累计）：']
    for mode,s in mode_summaries.items(): lines.append(f"- {mode}: `{json.dumps(s['all_conflicts'],ensure_ascii=False)}`")
    lines+=['','每次额外模型调用/预算跳过：']
    for mode,s in mode_summaries.items():
        for r in s['runs']: lines.append(f"- {mode}/{r['repeat']}: {r['extra_model_calls']} / {r['budget_skip_count']}；证据 `{r['evidence']}/summary.json`。")
    lines+=['','## llm_only 危险、hybrid 无危险的逐题对照','',
        '同重复编号配对。两组模型提议可能不同；不能仅凭同题结果差异声称校验因果。下表单列 hybrid 自身原始提议经只保留 V1/V10/目录检查的确定性离线复算也危险的题；复算无模型调用，不计为新增评测或延迟样本。完整提议、分数和 SHA256 见 `evaluation/results/intent_p3_A_analysis/`。',
        '', '| 题/重复 | hybrid 冲突码 | hybrid 完全正确 | 同一 hybrid 提议危险被去除 |','|---|---|---|---|']
    for v in observed_pairs: lines.append(f"| {v['case_id']}/{v['repeat']} | {', '.join(v['hybrid_conflicts']) or '无'} | {v['hybrid_fully_correct']} | {v['same_hybrid_raw_proposal_dangerous']} |")
    lines+=['','## 同一原始提议的校验效果','', '| 题/重复 | 移除危险 | 原本正确反而变错 | 冲突码 |','|---|---|---|---|']
    for v in validation_effects: lines.append(f"| {v['case_id']}/{v['repeat']} | {v['dangerous_removed']} | {v['validation_worsened']} | {', '.join(v['conflicts'])} |")
    if not validation_effects: lines.append('| 无 | — | — | — |')
    lines+=['','所有耗时按 nearest-rank；各次成功状态 p50/p95 和缺测数保留于 sealed summary.json，未删除失败。合同补充说明：[SCORING_AMENDMENT_01](../docs/plans/SCORING_AMENDMENT_01.md)；未改 test gold。',
        '', '运行前证据：`evaluation/results/intent_p3_preflight/`。原669项离线测试通过；首次沙箱临时目录权限失败保留，改用授权工作区临时目录后通过。这是离线环境检查，正式 test 未重跑。']
    (ROOT/'evaluation/INTENT_P3_A.md').write_text('\n'.join(lines)+'\n',encoding='utf-8')
    print(json.dumps({m:{k:s[k] for k in ['mean_exact_accuracy','worst_exact_accuracy','worst_dangerous']} for m,s in mode_summaries.items()},ensure_ascii=False))
